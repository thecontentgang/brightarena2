            className="bg-[#ff7043] text-white px-7 py-4 rounded-2xl text-xs font-bold tracking-widest uppercase shadow-lg shadow-[#ff7043]/20 text-center transition-colors hover:bg-[#ffc107] hover:text-[#4a1c13] active:scale-95 touch-manipulation"
          >
            Book an Interior Design Consultation
          </button>
        </div>
      </section>

      <ProjectModal isOpen={isModalOpen} onClose={() => setIsModalOpen(false)} />
    </div>
  );
};
